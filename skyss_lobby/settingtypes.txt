# Time of day kept in the hub, from 0 to 1 (0.5 = noon). -1 lets time run.
# Set time_speed = 0 as well, so the sun does not move between checks.
skyss_lobby_time_of_day (Fixed time of day) float 0.5 -1 1

# Keep the sky clear: no rain, snow or thunderstorms.
skyss_lobby_clear_weather (Always clear weather) bool true

# Player speed factor, walking and sprinting alike (1 = normal).
skyss_lobby_speed (Player speed) float 1.3 0.5 3
